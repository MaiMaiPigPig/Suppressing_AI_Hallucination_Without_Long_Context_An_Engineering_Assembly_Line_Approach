### 2.2 各步骤文件详细接口

#### race.js

```javascript
import { character } from '../core/state/character-model.js';
import { dataCache } from '../core/data/cache.js';

/**
 * 渲染种族选择步骤
 * @param {HTMLElement} container
 */
export function renderRaceStep(container) {
  // 1. 从 dataCache.races 获取种族列表
  // 2. 渲染种族卡片
  // 3. 绑定选择事件 -> character.race = selectedId
}

/**
 * 获取种族步骤摘要
 * @returns {{name: string, value: string}|null}
 */
export function getRaceStepSummary() {
  if (!character.race) return null;
  const race = dataCache.races.find(r => r.id === character.race);
  return { name: '种族', value: race?.name || '未知' };
}
```
