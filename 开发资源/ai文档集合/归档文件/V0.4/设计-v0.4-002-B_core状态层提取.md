```javascript
import { character } from './character-model.js';
import { STEP_DATA_FIELDS } from '../constants/character-constants.js';

/**
 * unbind 开关（控制某些步骤的数据绑定行为）
 * 原位置：creation/character.js L123-L128
 */

/**
 * 重置 unbind 状态
 */
export function resetUnbind() {
  unbind.skill = false;
  unbind.bonus = false;
  unbind.feat = false;
}

/**
 * 清除指定步骤之后的数据
 * 原位置：creation/character.js L68-L90 的 STEP_DATA_FIELDS 定义 + 清除逻辑
 * 
 * 当用户回退到某步骤时，清除后续步骤已填的数据，避免脏数据
 * 
 * @param {number} currentStep - 当前步骤编号
 * @param {string} mode - 模式：'creation' 或 'upgrade'，默认 'creation'
 */
export function clearDataAfterStep(currentStep, mode = 'creation') {
  const mapping = mode === 'upgrade' ? UPGRADE_STEP_DATA_FIELDS : STEP_DATA_FIELDS;
  const maxStep = mode === 'upgrade' ? 6 : 17;
  for (let step = currentStep + 1; step <= maxStep; step++) {
    const fields = mapping[step];
    if (fields) {
      for (const field of fields) {
        if (field in character) {
          character[field] = getDefaultValue(field);
        }
      }
    }
  }
}

/**
 * 获取字段默认值
 * @param {string} field - 字段名
 * @returns {*} 默认值
 */
function getDefaultValue(field) {
  // 根据字段类型返回默认值
  // 数组 -> [], 对象 -> {}, 字符串 -> '', 其他 -> null
}
```

## 六、风险与回滚

| 风险 | 应对方案 |
|------|----------|
| character 对象字段遗漏 | 对比新旧 `JSON.stringify(character)`，确保字段一致 |
| clearDataAfterStep 遗漏字段 | 对比原 STEP_DATA_FIELDS 和新的，确保映射完整 |
| steps/*.js 未更新 ABILITY_NAMES 引用 | 全局搜索 `ABILITY_NAMES`，确认所有引用点已更新 import |
| upgrade.js 的 character 默认值不同 | 对比 upgrade.js L30-L73 和 character.js L6-L35，取并集 |

**回滚方案**：保留原文件备份，恢复全局变量定义即可。
