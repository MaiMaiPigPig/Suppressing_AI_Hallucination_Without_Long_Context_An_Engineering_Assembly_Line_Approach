import { parseProficiencyLink } from './proficiency-link-parser.js';

/**
 * 获取特性中的熟练项数值
 * 原位置：creation/steps/sheet.js L863-L897（getFeatureProficiencyValues）
 * 
 * 遍历角色的所有特性，解析其中的 proficiency_link，
 * 返回每个熟练项的累计数值。
 * 
 * @param {Object} character - 角色对象
 * @returns {Object} 熟练项数值映射 {proficiencyId: value}
 */
export function getFeatureProficiencyValues(character) {
  const values = {};
  
  // 遍历角色的所有特性
  for (const feature of character.features || []) {
    if (feature.proficiency_link) {
      const parsed = parseProficiencyLink(feature.proficiency_link);
      const proficiency = findProficiencyInCache(parsed);
      if (proficiency) {
        values[proficiency.id] = (values[proficiency.id] || 0) + (parsed.value || 1);
      }
    }
  }
  
  return values;
}
