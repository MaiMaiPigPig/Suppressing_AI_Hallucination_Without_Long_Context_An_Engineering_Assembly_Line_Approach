## 一、模块划分

### 1.1 文件结构

```
creation/flow/
├── lifecycle.js      # 生命周期：欢迎页、新建、导入
├── step-router.js    # 步骤路由：17 步分发
└── navigation.js     # 导航控制：前进、后退、跳转、校验
```

### 1.2 职责边界

| 文件 | 职责 | 不职责 |
|------|------|--------|
| lifecycle.js | 应用入口、欢迎页渲染、角色新建/导入 | 不管理步骤切换 |
| step-router.js | 根据步骤编号调用对应渲染函数 | 不处理用户导航事件 |
| navigation.js | 处理 prev/next/goTo 点击、步骤校验、进度更新 | 不渲染具体步骤内容 |

```javascript
/**
 * 渲染指定步骤
 * 原位置：creation/character.js L535
 * 
 * @param {number} step - 步骤编号（1-17）
 * @param {HTMLElement} container - 渲染容器（可选，默认 #step-content）
 * @returns {Promise<void>}
 * @throws {Error} 步骤文件加载失败时抛出
 */
export async function renderStep(step, container = document.getElementById('step-content')) {
  const stepInfo = STEP_RENDERERS[step];
  if (!stepInfo) {
    console.error(`未知步骤: ${step}`);
    return;
  }
  
  // 检查步骤是否可跳过（法术步骤）
  if (stepInfo.skippable && isSpellStepSkippable(character)) {
    // 自动跳到下一步
    return;
  }
  
  // 动态导入并渲染
  // 异常策略：import 失败时抛出错误，由 navigation.js 捕获并显示提示
  const renderFn = await stepInfo.render();
  renderFn(container);
}
```

### 2.3 navigation.js

```javascript
import { character } from '../core/state/character-model.js';
import { clearDataAfterStep } from '../core/state/character-state.js';
import { renderStep } from './step-router.js';
import { updateProgress } from '../ui/progress-bar.js';
import { updateButtons } from '../ui/progress-bar.js'; // 或独立文件

/**
 * 当前步骤（运行时状态）
 * 原位置：creation/character.js L120
 */
export let currentStep = 1;

/**
 * 设置当前步骤（内部使用）
 * @param {number} step
 */
function setCurrentStep(step) {
  currentStep = step;
}

/**
 * 下一步
 * 原位置：creation/character.js L717
 */
export async function nextStep() {
  // 1. 校验当前步骤
  if (!validateStep(currentStep)) {
    return;
  }
  
  // 2. 计算下一步
  const next = currentStep + 1;
  
  // 3. 渲染下一步（带异常捕获）
  setCurrentStep(next);
  try {
    await renderStep(next);
    updateProgress(next);
    updateButtons(next);
  } catch (err) {
    console.error(`步骤 ${next} 渲染失败:`, err);
    // 回退到原步骤
    setCurrentStep(currentStep);
    showErrorToast(`步骤加载失败，请重试`);
  }
}

/**
 * 上一步
 * 原位置：creation/character.js L796
 */
export async function prevStep() {
  // 1. 清除当前步骤之后的数据
  clearDataAfterStep(currentStep - 1);
  
  // 2. 计算上一步
  const prev = currentStep - 1;
  
  // 3. 渲染上一步（带异常捕获）
  setCurrentStep(prev);
  try {
    await renderStep(prev);
    updateProgress(prev);
    updateButtons(prev);
  } catch (err) {
    console.error(`步骤 ${prev} 渲染失败:`, err);
    showErrorToast(`步骤加载失败，请重试`);
  }
}

/**
 * 跳转到指定步骤
 * 原位置：creation/character.js L829
 * 
 * @param {number} step - 目标步骤编号
 */
export async function goToStep(step) {
  // 1. 校验是否可以跳转（如未选种族不能跳到技能）
  if (!canNavigateTo(step)) {
    return;
  }
  
  // 2. 清除目标步骤之后的数据
  clearDataAfterStep(step - 1);
  
  // 3. 跳转（带异常捕获）
  setCurrentStep(step);
  try {
    await renderStep(step);
    updateProgress(step);
    updateButtons(step);
  } catch (err) {
    console.error(`步骤 ${step} 渲染失败:`, err);
    setCurrentStep(currentStep);
    showErrorToast(`步骤加载失败，请重试`);
  }
}

/**
 * 校验当前步骤是否完成
 * @param {number} step - 步骤编号
 * @returns {boolean} 是否通过校验
 */
function validateStep(step) {
  // 根据步骤类型进行不同校验
  // 如：种族步骤校验 character.race 是否已选
  // 技能步骤校验 classSkills 是否已选满
}

/**
 * 检查是否可以导航到目标步骤
 * @param {number} targetStep - 目标步骤
```

### 3.2 navigation.js 事件绑定

```javascript
/**
 * 初始化导航事件绑定
 * 在 DOM Ready 时调用
 */
export function initNavigation() {
  // 进度条点击事件委托
  document.querySelector('.progress-bar').addEventListener('click', (e) => {
    const stepEl = e.target.closest('[data-step]');
    if (stepEl) {
      const step = parseInt(stepEl.dataset.step, 10);
      goToStep(step);
    }
  });
  
  // 上一步按钮
  document.getElementById('btn-prev').addEventListener('click', prevStep);
  
  // 下一步按钮
  document.getElementById('btn-next').addEventListener('click', nextStep);
}

/**
 * 显示错误提示（简单实现，可替换为 toast 组件）
 * @param {string} message
 */
function showErrorToast(message) {
  // v0.4 边界：使用 alert 或 console，不引入新 UI 组件
  alert(message);
}
```

## 七、风险与回滚

| 风险 | 应对方案 |
|------|----------|
| 动态 import 不支持 | 如环境不支持，改为静态 import 所有步骤文件 |
| 事件委托绑定失败 | 确保 DOM Ready 后再绑定，或使用 jQuery $(document).ready |
| 步骤校验逻辑遗漏 | 对比原 character.js 的校验逻辑，确保全部迁移 |
| currentStep 状态丢失 | navigation.js 导出 currentStep，其他模块只读不直接修改 |

**回滚方案**：恢复 character.html 的 onclick 属性，恢复 character.js 的导航函数。
