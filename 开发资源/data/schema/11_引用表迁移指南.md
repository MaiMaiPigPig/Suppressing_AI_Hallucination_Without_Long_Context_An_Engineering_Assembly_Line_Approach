# 11_引用表迁移指南

## 背景

v1.0.0 版本废除了引用表（中间表），改用自描述表达式。本文档指导开发者如何迁移现有数据。

---

## 迁移范围

### 需要修改的文件

1. **所有引用 `link(id)` 的 CSV 文件**：
   - `csv导出/职业特性.csv` - 特性规则字段
   - `csv导出/种族特性.csv` - 特性规则字段
   - `csv导出/背景.csv` - 技能、装备、熟练字段
   - `csv导出/专长.csv` - 车卡数据字段
   - `csv导出/职业.csv` - 武器/护甲/工具熟练字段
   - `csv导出/职业额外资源.csv` - 资源规则字段

2. **转换脚本**：
   - `csv转json.py` - 新增自描述表达式解析

3. **前端代码**：
   - `TableChronicle/static/character/dnd_5r/creation/handlers/link.js` - 兼容新旧语法

## 迁移步骤

### 步骤 1：生成映射表

运行以下脚本，将现有引用表转换为自描述表达式：

```python
# migrate_link.py
import csv
import json

# 读取引用表
links = {}
with open('csv导出/引用表.csv', 'r', encoding='utf-8-sig') as f:
    reader = csv.DictReader(f)
    for row in reader:
        links[row['id']] = row

# 转换为自描述表达式
def convert_link(link_id):
    link = links.get(str(link_id))
    if not link:
        return f"UNKNOWN_LINK({link_id})"
    
    source_table = link['source_table']
    mode = link['mode']
    type_ = link['type']
    target_id = link['target_id']
    count = link['count']
    complex_logic = link.get('complex_logic', '')
    
    # 表名映射
    TABLE_MAP = {
        'feats': 'feats',
        'spells': 'spells',
        'skills': 'skills',
        'tools': 'tools',
        'weapons': 'weapons',
        'armors': 'armors',
        'languages': 'languages',
        'packages': 'packages',
        'growup': 'growup',
        'spell_slot': 'spell_slot',
        'basic': ''  # 基础规则，不需要表名
    }
    
    table = TABLE_MAP.get(source_table, source_table)
    
    # type=id → 直接引用
    if type_ == 'id':
        if table:
            return f"{table}(id={target_id})"
        return f"id={target_id}"
    
    # type=type → 按类型筛选
    if type_ == 'type':
        if count:
            return f"{table}(type={target_id})+{count}"
        return f"{table}(type={target_id})"
    
    # type=filter → 条件筛选
    if type_ == 'filter':
        if count:
            return f"{table}({complex_logic})+{count}"
        return f"{table}({complex_logic})"
    
    # type=choose → 手动列表
    if type_ == 'choose':
        return f"choose({table}({complex_logic}))+{count}"
    
    # type=special → 特殊操作
    if type_ == 'special':
        if complex_logic == 'all':
            return f"all({table})"
        if complex_logic == 'half-prof':
            return f"half-prof({table})"
        if complex_logic.startswith('proficiency>pure'):
            if '(' in complex_logic:
                ids = complex_logic.split('(')[1].split(')')[0]
                return f"proficiency>pure({ids})+{count}"
            return f"proficiency>pure+{count}"
        if complex_logic.startswith('choose('):
            # choose(tool(type=66),tool(type=67))
            return complex_logic.replace('tool(', 'tools(').replace('skill(', 'skills(') + f"+{count}"
        return complex_logic
    
    # type=spell_slot → 法术位
    if type_ == 'spell_slot':
        return f"spell_slot(lv={target_id})+{count}"
    
    # type=prerequisite → 前置条件
    if type_ == 'prerequisite':
        return f"prerequisite({complex_logic})"
    
    # type=rules → 规则表达式
    if type_ == 'rules':
        return complex_logic
    
    return f"UNKNOWN_TYPE({type_},{link_id})"

# 测试
for link_id in [1, 58, 106, 257, 187, 296, 306, 506, 514]:
    print(f"link({link_id}) → {convert_link(link_id)}")
```

### 步骤 2：批量替换 CSV 中的 link(id)

```python
# replace_links.py
import csv
import re

# 读取映射表（从步骤1生成）
link_map = {}
with open('link_mapping.csv', 'r') as f:
    reader = csv.DictReader(f)
    for row in reader:
        link_map[row['old']] = row['new']

# 需要处理的文件
files_to_process = [
    'csv导出/职业特性.csv',
    'csv导出/种族特性.csv',
    'csv导出/背景.csv',
    'csv导出/专长.csv',
    'csv导出/职业.csv',
    'csv导出/职业额外资源.csv'
]

# 替换函数
def replace_links_in_text(text):
    if not text:
        return text
    
    # 匹配 link(id)
    pattern = r'link\((\d+)\)'
    
    def replace_match(match):
        link_id = match.group(1)
        old_expr = f"link({link_id})"
        new_expr = link_map.get(old_expr, old_expr)
        return new_expr
    
    return re.sub(pattern, replace_match, text)

# 处理每个文件
for filepath in files_to_process:
    with open(filepath, 'r', encoding='utf-8-sig') as f:
        reader = csv.DictReader(f)
        fieldnames = reader.fieldnames
        rows = list(reader)
    
    # 替换所有字段中的 link(id)
    for row in rows:
        for key in row:
            if row[key]:
                row[key] = replace_links_in_text(row[key])
    
    # 写回
    with open(filepath, 'w', encoding='utf-8', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(rows)
    
    print(f"已处理: {filepath}")
```

### 步骤 3：更新转换脚本

在 `csv转json.py` 中新增自描述表达式解析：

```python
# 在 csv转json.py 中新增

import re

class ExpressionParser:
    """自描述表达式解析器"""
    
    # 基本引用: 表名(条件)+数量
    BASIC_REF = re.compile(r'^(\w+)\(([^)]+)\)(?:\+(\d+))?$')
    
    # 手动选择: choose(表名(id1,id2))+数量
    CHOOSE_REF = re.compile(r'^choose\((\w+)\(([^)]+)\)\)(?:\+(\d+))?$')
    
    # 范围选择: 表名(id→id)
    RANGE_REF = re.compile(r'^(\w+)\((\d+)→(\d+)\)$')
    
    # 特殊操作: all(表名) / half-prof(表名) / spell_slot(lv=N)+M
    SPECIAL_REF = re.compile(r'^(all|half-prof)\((\w+)\)$')
    SPELL_SLOT_REF = re.compile(r'^spell_slot\(lv=(\d+)\)\+(\d+)$')
    PROFICIENCY_REF = re.compile(r'^proficiency>pure(?:\(([^)]+)\))?\+(\d+)$')
    
    # 属性调整值
    ABILITY_MOD = re.compile(r'AbilityModifiers\((\d+)\)')
    LV_REF = re.compile(r'lv:class\((\d+)\)')
    
    def __init__(self, data_dir='data/json/'):
        self.data_dir = data_dir
        self.cache = {}
    
    def parse(self, expr):
        """解析表达式，返回解析结果"""
        
        # 过渡期内支持旧 link 语法
        if expr.startswith('link('):
            return self._parse_legacy_link(expr)
        
        # 基本引用
        match = self.BASIC_REF.match(expr)
        if match:
            return self._parse_basic(match)
        
        # 手动选择
        match = self.CHOOSE_REF.match(expr)
        if match:
            return self._parse_choose(match)
        
        # 范围选择
        match = self.RANGE_REF.match(expr)
        if match:
            return self._parse_range(match)
        
        # 特殊操作
        match = self.SPECIAL_REF.match(expr)
        if match:
            return self._parse_special(match)
        
        match = self.SPELL_SLOT_REF.match(expr)
        if match:
            return self._parse_spell_slot(match)
        
        match = self.PROFICIENCY_REF.match(expr)
        if match:
            return self._parse_proficiency(match)
        
        # 公式
        if 'AbilityModifiers' in expr or 'lv:class' in expr or 'Proficiency' in expr:
            return self._parse_formula(expr)
        
        # 多选一
        if expr.startswith('①'):
            return self._parse_multi_choice(expr)
        
        # 前置条件
        if expr.startswith('prerequisite('):
            return self._parse_prerequisite(expr)
        
        # 连锁引用
        if expr.startswith('chain:'):
            return self._parse_chain(expr)
        
        # 代码标记
        if expr.startswith('code('):
            return self._parse_code(expr)
        
        return {'type': 'unknown', 'expr': expr}
    
    def _parse_basic(self, match):
        table, conditions, count = match.groups()
        return {
            'type': 'basic_ref',
            'table': table,
            'conditions': self._parse_conditions(conditions),
            'count': int(count) if count else 1
        }
    
    def _parse_conditions(self, conditions_str):
        """解析条件字符串 'key1=value1,key2=value2'"""
        conditions = {}
        parts = conditions_str.split(',')
        
        for part in parts:
            if '=' in part:
                key, value = part.split('=', 1)
                key = key.strip()
                value = value.strip()
                
                # 同一 key 多次出现 = OR
                if key in conditions:
                    if not isinstance(conditions[key], list):
                        conditions[key] = [conditions[key]]
                    conditions[key].append(value)
                else:
                    conditions[key] = value
        
        return conditions
    
    def _parse_choose(self, match):
        table, ids, count = match.groups()
        return {
            'type': 'choose',
            'table': table,
            'ids': ids.split(','),
            'count': int(count) if count else 1
        }
    
    def _parse_range(self, match):
        table, start, end = match.groups()
        return {
            'type': 'range',
            'table': table,
            'start': int(start),
            'end': int(end)
        }
    
    def _parse_special(self, match):
        op, table = match.groups()
        return {
            'type': 'special',
            'operation': op,
            'table': table
        }
    
    def _parse_spell_slot(self, match):
        level, count = match.groups()
        return {
            'type': 'spell_slot',
            'level': int(level),
            'count': int(count)
        }
    
    def _parse_proficiency(self, match):
        ids, count = match.groups()
        return {
            'type': 'proficiency_upgrade',
            'ids': ids.split(',') if ids else [],
            'count': int(count)
        }
    
    def _parse_formula(self, expr):
        return {
            'type': 'formula',
            'expr': expr
        }
    
    def _parse_multi_choice(self, expr):
        choices = []
        parts = expr.split(',')
        for part in parts:
            if part.startswith('①') or part.startswith('②') or part.startswith('③'):
                choices.append(part[1:])
        return {
            'type': 'multi_choice',
            'choices': choices
        }
    
    def _parse_prerequisite(self, expr):
        condition = expr[13:-1]  # 去掉 prerequisite( 和 )
        return {
            'type': 'prerequisite',
            'condition': condition
        }
    
    def _parse_chain(self, expr):
        target_id = expr[6:]  # 去掉 chain:
        return {
            'type': 'chain',
            'target_id': int(target_id)
        }
    
    def _parse_code(self, expr):
        code = expr[5:-1]  # 去掉 code( 和 )
        return {
            'type': 'code',
            'code': code
        }
    
    def _parse_legacy_link(self, expr):
        """过渡期内支持旧 link 语法"""
        link_id = int(re.search(r'link\((\d+)\)', expr).group(1))
        # 查 legacy_lookup_table（从引用表生成）
        return {
            'type': 'legacy_link',
            'link_id': link_id,
            'note': '请迁移到自描述表达式'
        }


# 在 csv转json.py 中使用
parser = ExpressionParser()

# 示例：解析职业特性的 rules 字段
# rules = "spells(class=3,lv=0)+2"
# result = parser.parse(rules)
# result = {'type': 'basic_ref', 'table': 'spells', 'conditions': {'class': '3', 'lv': '0'}, 'count': 2}
```

### 步骤 4：更新前端 link.js

```javascript
// TableChronicle/static/character/dnd_5r/creation/handlers/link.js

class ExpressionParser {
    constructor() {
        // 正则表达式
        this.patterns = {
            basic: /^(\w+)\(([^)]+)\)(?:\+(\d+))?$/,
            choose: /^choose\((\w+)\(([^)]+)\)\)(?:\+(\d+))?$/,
            range: /^(\w+)\((\d+)→(\d+)\)$/,
            special: /^(all|half-prof)\((\w+)\)$/,
            spellSlot: /^spell_slot\(lv=(\d+)\)\+(\d+)$/,
            proficiency: /^proficiency>pure(?:\(([^)]+)\))?\+(\d+)$/,
            abilityMod: /AbilityModifiers\((\d+)\)/g,
            lvRef: /lv:class\((\d+)\)/g,
            legacyLink: /^link\((\d+)\)$/
        };
        
        // 过渡期内：旧 link 映射表（从引用表生成）
        this.legacyMap = {
            'link(1)': 'feats(id=5)',
            'link(58)': 'weapons(type=63)',
            'link(106)': 'spells(id=56)',
            // ... 更多映射
        };
    }
    
    parse(expr) {
        if (!expr) return null;
        
        // 过渡期内支持旧 link 语法
        if (expr.startsWith('link(')) {
            const newExpr = this.legacyMap[expr];
            if (newExpr) {
                console.warn(`[Legacy] ${expr} → ${newExpr}`);
                expr = newExpr;
            } else {
                console.error(`[Legacy] Unknown link: ${expr}`);
                return null;
            }
        }
        
        // 基本引用
        let match = expr.match(this.patterns.basic);
        if (match) {
            return this.parseBasic(match);
        }
        
        // 手动选择
        match = expr.match(this.patterns.choose);
        if (match) {
            return this.parseChoose(match);
        }
        
        // 范围选择
        match = expr.match(this.patterns.range);
        if (match) {
            return this.parseRange(match);
        }
        
        // 特殊操作
        match = expr.match(this.patterns.special);
        if (match) {
            return this.parseSpecial(match);
        }
        
        // 法术位
        match = expr.match(this.patterns.spellSlot);
        if (match) {
            return this.parseSpellSlot(match);
        }
        
        // 专精
        match = expr.match(this.patterns.proficiency);
        if (match) {
            return this.parseProficiency(match);
        }
        
        // 公式
        if (expr.includes('AbilityModifiers') || expr.includes('lv:class') || expr.includes('Proficiency')) {
            return this.parseFormula(expr);
        }
        
        // 多选一
        if (expr.startsWith('①')) {
            return this.parseMultiChoice(expr);
        }
        
        // 前置条件
        if (expr.startsWith('prerequisite(')) {
            return this.parsePrerequisite(expr);
        }
        
        // 连锁引用
        if (expr.startsWith('chain:')) {
            return this.parseChain(expr);
        }
        
        // 代码标记
        if (expr.startsWith('code(')) {
            return this.parseCode(expr);
        }
        
        return { type: 'unknown', expr };
    }
    
    parseBasic(match) {
        const [, table, conditions, count] = match;
        return {
            type: 'basic',
            table,
            conditions: this.parseConditions(conditions),
            count: parseInt(count) || 1
        };
    }
    
    parseConditions(conditionsStr) {
        const conditions = {};
        const parts = conditionsStr.split(',');
        
        for (const part of parts) {
            const [key, value] = part.split('=').map(s => s.trim());
            if (key && value !== undefined) {
                if (conditions[key]) {
                    if (!Array.isArray(conditions[key])) {
                        conditions[key] = [conditions[key]];
                    }
                    conditions[key].push(value);
                } else {
                    conditions[key] = value;
                }
            }
        }
        
        return conditions;
    }
    
    parseChoose(match) {
        const [, table, ids, count] = match;
        return {
            type: 'choose',
            table,
            ids: ids.split(','),
            count: parseInt(count) || 1
        };
    }
    
    parseRange(match) {
        const [, table, start, end] = match;
        return {
            type: 'range',
            table,
            start: parseInt(start),
            end: parseInt(end)
        };
    }
    
    parseSpecial(match) {
        const [, operation, table] = match;
        return {
            type: 'special',
            operation,
            table
        };
    }
    
    parseSpellSlot(match) {
        const [, level, count] = match;
        return {
            type: 'spell_slot',
            level: parseInt(level),
            count: parseInt(count)
        };
    }
    
    parseProficiency(match) {
        const [, ids, count] = match;
        return {
            type: 'proficiency',
            ids: ids ? ids.split(',') : [],
            count: parseInt(count)
        };
    }
    
    parseFormula(expr) {
        return {
            type: 'formula',
            expr
        };
    }
    
    parseMultiChoice(expr) {
        const choices = [];
        const parts = expr.split(',');
        for (const part of parts) {
            if (part.startsWith('①') || part.startsWith('②') || part.startsWith('③')) {
                choices.push(part.substring(1));
            }
        }
        return {
            type: 'multi_choice',
            choices
        };
    }
    
    parsePrerequisite(expr) {
        const condition = expr.slice(13, -1);
        return {
            type: 'prerequisite',
            condition
        };
    }
    
    parseChain(expr) {
        const targetId = expr.slice(6);
        return {
            type: 'chain',
            targetId: parseInt(targetId)
        };
    }
    
    parseCode(expr) {
        const code = expr.slice(5, -1);
        return {
            type: 'code',
            code
        };
    }
}

// 导出
window.ExpressionParser = ExpressionParser;
```
